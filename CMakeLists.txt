cmake_minimum_required(VERSION 3.21)
project(unishare VERSION 0.1.0 LANGUAGES CXX)
include(cmake/Dependencies.cmake)
add_library(unishare_core STATIC src/core/device_catalog.cpp src/protocol/frame_codec.cpp src/protocol/wire_codec.cpp)
add_library(unishare::core ALIAS unishare_core)
target_compile_features(unishare_core PUBLIC cxx_std_17)
target_include_directories(unishare_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
# External dependency links are added by their first consuming implementation task.
include(CTest)
if(BUILD_TESTING)
 add_executable(protocol_test tests/core/protocol_test.cpp)
 target_include_directories(protocol_test PRIVATE src)
 target_link_libraries(protocol_test PRIVATE unishare::core)
 add_test(NAME protocol COMMAND protocol_test)
endif()
if(BUILD_TESTING)
 add_executable(contracts_test tests/core/contracts_test.cpp)
 target_link_libraries(contracts_test PRIVATE unishare::core)
 add_test(NAME contracts COMMAND contracts_test)
endif()
if(BUILD_TESTING)
 add_executable(device_catalog_test tests/core/device_catalog_test.cpp)
 target_include_directories(device_catalog_test PRIVATE src)
 target_link_libraries(device_catalog_test PRIVATE unishare::core)
 add_test(NAME device_catalog COMMAND device_catalog_test)
endif()
