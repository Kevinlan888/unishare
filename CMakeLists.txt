cmake_minimum_required(VERSION 3.21)
project(unishare VERSION 0.1.0 LANGUAGES CXX)
include(cmake/Dependencies.cmake)
add_library(unishare_core STATIC src/core/transfer_engine.cpp src/core/session.cpp src/core/pairing.cpp src/core/identity.cpp src/core/device_catalog.cpp src/protocol/frame_codec.cpp src/protocol/wire_codec.cpp)
add_library(unishare::core ALIAS unishare_core)
target_sources(unishare_core PRIVATE src/core/share_sdk.cpp src/core/serial_scheduler.cpp src/core/operation_registry.cpp)
target_sources(unishare_core PRIVATE src/core/connection_service.cpp src/core/transfer_service.cpp src/core/clipboard_service.cpp)
target_compile_features(unishare_core PUBLIC cxx_std_17)
if(MSVC)
 target_compile_options(unishare_core PRIVATE /permissive-)
endif()
target_include_directories(unishare_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
unishare_require_openssl()
target_link_libraries(unishare_core PUBLIC OpenSSL::SSL OpenSSL::Crypto)
find_package(Threads REQUIRED)
target_link_libraries(unishare_core PUBLIC Threads::Threads)
include(CTest)

 add_executable(sdk_services_test tests/core/sdk_services_test.cpp)
 target_include_directories(sdk_services_test PRIVATE src)
 target_link_libraries(sdk_services_test PRIVATE unishare::core)
 unishare_stage_openssl_runtime(sdk_services_test)
 add_test(NAME sdk_services COMMAND sdk_services_test)
endif()
